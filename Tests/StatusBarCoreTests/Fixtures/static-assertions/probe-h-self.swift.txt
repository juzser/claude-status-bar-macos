// Fixture (NEVER COMPILED): probe (h)'s own implementation, needle and
// diagnostic included. Probe (h) must PASS over its own source -- the
// needle constant and the diagnostic string are inert string data.
import Foundation
import Testing

enum FixtureProbeHSelf {
    static func run(_ files: [URL]) throws {
        let needle = "TrustedApplicationGate.defaultOnReject"
        for file in files {
            let text = try String(contentsOf: file, encoding: .utf8)
            #expect(text.contains(needle) == false,
                    "\(file.lastPathComponent) holds an executable reference to \(needle)")
        }
    }
}
